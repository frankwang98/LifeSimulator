cmake_minimum_required(VERSION 3.16)
project(life_tree LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(life_tree src/main.cpp)
include(CTest)
if(BUILD_TESTING)
  add_executable(life_tests tests/test_life.cpp)
  target_include_directories(life_tests PRIVATE src)
  add_test(NAME simulation COMMAND life_tests)
  add_test(NAME cli_help COMMAND life_tree --help)
  add_test(NAME cli_invalid COMMAND life_tree --days -1)
  set_tests_properties(cli_invalid PROPERTIES WILL_FAIL TRUE)
endif()
